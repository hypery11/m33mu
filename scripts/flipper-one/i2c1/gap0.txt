# I2C1 scripted master: back-to-back transfers.  `gap_us 0` puts a START at
# the same emulated time as the STOP before it (no bus free time at all,
# which a real bus does not allow: tBUF >= 1.3 us at 400 kHz).  Shows what
# the firmware does when the STOP of one transfer and the START of the next
# reach its ISR together.
wait_log enabling I2C slave mode
delay_ms 50
write_read 69 03 10 : 2            # 0x0310 LED link1 colour before: 00 00
gap_us 0
write_read 69 00 80 : 2            # version, 01 00
write 69 03 10 78 56               # 0x0310 := 0x5678, START at the time of the STOP above
gap_us 50
delay_ms 20
write_read 69 03 10 : 2            # 78 56 if the write was taken
write_read 69 00 80 : 2            # 01 00 unless left-over bytes are taken as the register address
write_read 69 00 80 : 2
write_read 69 00 80 : 2
